Quick Start

Include the engine and execute instructions from any event listener:

import { runInstructions } from "dom-primitive-engine";
// Or in browser: window.ks.domPrimitiveEngine.runInstructions(...)

button.addEventListener("click", (event) => {
  const rowData = runInstructions(event.currentTarget, instructions, {
    columns: ["itemName", "category", "rate"]
  });
  console.log("Row Data:", rowData);
});
